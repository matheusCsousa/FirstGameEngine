cmake_minimum_required(VERSION 3.24)
project(fersa VERSION 1.0 LANGUAGES C CXX)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(FERSA_FETCH_DEPENDENCIES "Fetch versioned GLFW, GLEW and GLM sources" ON)
include(cmake/Dependencies.cmake)
add_subdirectory(core)
add_subdirectory(gameApp)
option(FERSA_BUILD_EDITOR "Build the visual scene editor" ON)
if(FERSA_BUILD_EDITOR)
  add_subdirectory(editor)
endif()
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
# Portable editor package: preserve assets beside the executable.
if(FERSA_BUILD_EDITOR)
  install(TARGETS fersa_editor RUNTIME DESTINATION . COMPONENT Editor)
  install(DIRECTORY gameApp/models gameApp/shader gameApp/textures DESTINATION assets COMPONENT Editor)
  install(FILES RELEASE-README.txt DESTINATION . COMPONENT Editor)
  install(DIRECTORY editor/licenses/ DESTINATION licenses COMPONENT Editor)
  foreach(dependency imgui imguizmo glfw glm glew_source nlohmann_json)
    FetchContent_GetProperties(${dependency} SOURCE_DIR notice_source)
    if(dependency STREQUAL "imgui" AND FERSA_IMGUI_SOURCE_DIR)
      set(notice_source "${FERSA_IMGUI_SOURCE_DIR}")
    endif()
    if(notice_source)
      file(GLOB notices "${notice_source}/LICENSE*" "${notice_source}/COPYING*" "${notice_source}/copying*" "${notice_source}/readme.txt")
      if(notices)
        install(FILES ${notices} DESTINATION "licenses/${dependency}" COMPONENT Editor)
      endif()
    endif()
  endforeach()
  if(MSVC)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
    set(CMAKE_INSTALL_SYSTEM_RUNTIME_COMPONENT Editor)
    include(InstallRequiredSystemLibraries)
  endif()
  set(CPACK_INSTALL_CMAKE_PROJECTS "${CMAKE_BINARY_DIR};fersa;Editor;/")
  set(CPACK_GENERATOR ZIP)
  set(CPACK_PACKAGE_NAME Fersa-Editor)
  set(CPACK_PACKAGE_VERSION 0.2.0)
  set(package_arch "${CMAKE_SYSTEM_PROCESSOR}")
  if(NOT package_arch AND WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(package_arch x64)
  endif()
  set(CPACK_PACKAGE_FILE_NAME "Fersa-Editor-0.2.0-${CMAKE_SYSTEM_NAME}-${package_arch}")
  include(CPack)
endif()
